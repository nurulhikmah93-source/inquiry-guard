    let audioEnabled = true;

    // AUDIO SYNTHESIZER (Web Audio API)
    function playSound(type) {
        if (!audioEnabled) return;
        try {
            const ctx = new (window.AudioContext || window.webkitAudioContext)();
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.connect(gain);
            gain.connect(ctx.destination);

            if (type === 'correct') {
                osc.frequency.setValueAtTime(523.25, ctx.currentTime); // C5
                osc.frequency.setValueAtTime(659.25, ctx.currentTime + 0.1); // E5
                gain.gain.fadeOut(ctx.currentTime + 0.3);
                osc.start();
                osc.stop(ctx.currentTime + 0.3);
            } else if (type === 'wrong') {
                osc.frequency.setValueAtTime(220, ctx.currentTime); // A3
                osc.frequency.setValueAtTime(196, ctx.currentTime + 0.15); // G3
                gain.gain.fadeOut(ctx.currentTime + 0.4);
                osc.start();
                osc.stop(ctx.currentTime + 0.4);
            }
        } catch(e) {
            console.log("Audio Error/Not Supported");
        }
    }

    function toggleAudio() {
        audioEnabled = !audioEnabled;
        const btn = document.getElementById('audio-toggle');
        btn.innerText = audioEnabled ? "🔊 Suara: ON" : "🔇 Suara: OFF";
    }

    // UTILS NAVIGATION
    function showScreen(screenId) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }

    function startGame() {
        currentQ = 0;
        score = 0;
        correctAns = 0;
        wrongAns = 0;
        showScreen('screen-quiz');
        loadQuestion();
    }

    function loadQuestion() {
        const qData = questions[currentQ];
        document.getElementById('question-number').innerText = `Soal ${currentQ + 1} / ${questions.length}`;
        document.getElementById('score-display').innerText = `Skor: ${score}`;
        document.getElementById('progress-bar').style.width = `${((currentQ) / questions.length) * 100}%`;
        
        document.getElementById('question-text').innerText = qData.q;
        
        const optContainer = document.getElementById('options-container');
        optContainer.innerHTML = '';

        qData.options.forEach((opt, index) => {
            const btn = document.createElement('button');
            btn.className = 'option-btn';
            btn.innerText = `${String.fromCharCode(65 + index)}. ${opt}`;
            btn.onclick = () => checkAnswer(index);
            optContainer.appendChild(btn);
        });

        const fb = document.getElementById('feedback-box');
        fb.className = 'feedback-box';
        fb.style.display = 'none';

        document.getElementById('next-btn').style.display = 'none';
    }

    function checkAnswer(selectedIndex) {
        const qData = questions[currentQ];
        const buttons = document.querySelectorAll('.option-btn');
        
        // Disable all buttons after choice
        buttons.forEach(btn => btn.disabled = true);

        const fb = document.getElementById('feedback-box');

        if (selectedIndex === qData.answer) {
            buttons[selectedIndex].classList.add('correct');
            score += 5;
            correctAns++;
            playSound('correct');
            fb.className = 'feedback-box show correct-fb';
            fb.innerHTML = `<strong>Jawaban Tepat! 🎉</strong><br>${qData.exp}`;
        } else {
            buttons[selectedIndex].classList.add('wrong');
            buttons[qData.answer].classList.add('correct');
            wrongAns++;
            playSound('wrong');
            fb.className = 'feedback-box show wrong-fb';
            fb.innerHTML = `<strong>Kurang Tepat! 💡</strong><br>${qData.exp}`;
        }

        document.getElementById('score-display').innerText = `Skor: ${score}`;
        document.getElementById('next-btn').style.display = 'block';
    }

    function nextQuestion() {
        currentQ++;
        if (currentQ < questions.length) {
            loadQuestion();
        } else {
            showResult();
        }
    }

    function showResult() {
        showScreen('screen-result');
        document.getElementById('final-score').innerText = score;
        document.getElementById('correct-count').innerText = correctAns;
        document.getElementById('wrong-count').innerText = wrongAns;

        let stars = "⭐";
        let badge = "Inquirer Pemula PAI";
        let motivation = "Terus tingkatkan pemahamanmu tentang hakikat salat dan zikir!";

            badge = "🏆 Lencana: Master Inquirer Guard PAI";
            motivation = "Luar biasa! Kamu sangat memahami bagaimana salat & zikir bentengi diri dari kemunkaran.";

            stars = "⭐⭐";
            badge = "🥇 Lencana: Penyelamat Karakter Remaja";
            motivation = "Bagus sekali! Pemahaman materi PAI-mu sudah sangat baik.";

        document.getElementById('motivation-text').innerText = motivation;
    }
